%----------------------------------------------------------------------------------------
%	5. WHY IT'S HARD / GAP IN PRIOR WORK
%----------------------------------------------------------------------------------------

\begin{frame}
	\frametitle{Why it is hard, and the two design choices that follow}

	\begin{columns}[T]
		\begin{column}{0.5\linewidth}
			\textbf{The task}
			\begin{itemize}
				\item No colour in NIR, so it is \textbf{one-to-many}: many RGBs fit one frame.
				\item Materials look different (foliage bright, water dark).
				\item Aligned pairs are scarce (camera parallax).
			\end{itemize}
		\end{column}
		\begin{column}{0.5\linewidth}
			\textbf{Prior work}
			\begin{itemize}
				\item GANs and diffusion, judged mainly on \textbf{image quality}.
				\item Diffusion is iterative and stochastic, wrong for a deterministic edge pre-processor.
			\end{itemize}
		\end{column}
	\end{columns}

	\vspace{0.6\baselineskip}
	\begin{beamercolorbox}[wd=0.96\linewidth, sep=9pt, rounded=true]{keymsg}
		\centering\textbf{Two design choices:} \;(1)~a deterministic, edge-friendly \textbf{restoration backbone};\quad (2)~supervise in \textbf{feature space}, not pixels.
	\end{beamercolorbox}
\end{frame}

\begin{frame}
	\frametitle{The backbone: an activation-free NAFNet}
	\framesubtitle{No activation function, so it quantises and exports cleanly}

	\vspace{0.8\baselineskip}
	\begin{columns}[T]
		\begin{column}{0.46\linewidth}
			\centering
			\vspace{0.3\baselineskip}
			\begin{tikzpicture}[
				font=\scriptsize,
				op/.style={rectangle, rounded corners, draw=ICLBlue, fill=ICLBlue!8, minimum width=2.4cm, minimum height=0.52cm, align=center},
				act/.style={rectangle, rounded corners, draw=BadRed, fill=BadRed!12, minimum width=2.4cm, minimum height=0.52cm, align=center},
				gate/.style={rectangle, rounded corners, draw=GoodGreen, fill=GoodGreen!16, minimum width=2.4cm, minimum height=0.52cm, align=center},
				ar/.style={-{Latex[length=1.4mm]}, ICLBlue}
			]
				\node[font=\scriptsize\bfseries] at (0,0.78) {Pix2Pix block};
				\node[op]  (a1) at (0,0.2)  {conv};
				\node[act] (a2) at (0,-0.42){ReLU / GELU};
				\node[op]  (a3) at (0,-1.04){conv};
				\draw[ar](a1)--(a2); \draw[ar](a2)--(a3);
				\node[font=\scriptsize\bfseries] at (3.1,0.78) {NAFNet block};
				\node[op]   (b1) at (3.1,0.2)  {conv};
				\node[gate] (b2) at (3.1,-0.42){SimpleGate};
				\node[op]   (b3) at (3.1,-1.04){conv};
				\draw[ar](b1)--(b2); \draw[ar](b2)--(b3);
			\end{tikzpicture}

			\vspace{0.4\baselineskip}
			{\footnotesize \textbf{SimpleGate:} split the channels in half, multiply the two halves. \up{First use of NAFNet for NIR$\to$RGB.}}
		\end{column}
		\begin{column}{0.5\linewidth}
			\textbf{Why drop the activation?}
			\begin{itemize}
				\setlength{\itemsep}{4pt}
				\item Keeps negatives, so no ``dead'' units and steadier training.
				\item The multiply gates one half with the other, capturing feature interactions.
				\item Predictable value range, so it \textbf{quantises to INT8 cleanly}.
				\item Small, standard op set, so \textbf{clean export} to ONNX, LiteRT, Hailo.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{The loss: supervise in feature space}
	\framesubtitle{Match the features frozen foundation models use, not the pixels}

	\centering
	\includegraphics[width=0.82\linewidth]{loss_diagram.png}

	\vspace{0.5\baselineskip}

		\begin{column}{0.52\linewidth}
			\begin{beamercolorbox}[wd=\linewidth, sep=8pt, rounded=true]{eqbox}
				\centering
				Train so that, for each frozen extractor $F$:
				\[ F\big(T(x_{\text{NIR}})\big)\;\approx\;F(x_{\text{RGB}}) \]

		\end{column}

%----------------------------------------------------------------------------------------
%	9. EVALUATION DESIGN: GAP CLOSURE
%----------------------------------------------------------------------------------------

\begin{frame}
	\frametitle{How I measure success: gap closure}
	\framesubtitle{Run a frozen model on three inputs and ask how much RGB behaviour is recovered}

	\centering
	\vspace{1.2\baselineskip}
	\begin{columns}
		\begin{column}{0.2\linewidth}\end{column}
		\begin{column}{0.6\linewidth}
			\begin{beamercolorbox}[wd=\linewidth, sep=10pt, rounded=true]{eqbox}
				\centering
				\[
					\text{gap closure} \;=\; \frac{m(\text{translated}) - m(\text{NIR})}{m(\text{RGB}) - m(\text{NIR})}
				\]
			\end{beamercolorbox}
		\end{column}
		\begin{column}{0.2\linewidth}\end{column}
	\end{columns}

	\vspace{1.0\baselineskip}
	\begin{columns}[T]
		\begin{column}{0.32\linewidth}
			\centering\textbf{\dn{0}}\\ no better than raw NIR
		\end{column}
		\begin{column}{0.32\linewidth}
			\centering\textbf{\up{1}}\\ native-RGB behaviour fully recovered
		\end{column}
		\begin{column}{0.32\linewidth}
			\centering\textbf{\dn{$<0$}}\\ translation made it \emph{worse}
		\end{column}
	\end{columns}

	\vspace{0.7\baselineskip}
	{\small Reference = the \emph{same} frozen model on native RGB (pseudo-labels), across detection, segmentation, depth and independent embedding probes. No human annotation required.}
\end{frame}

%----------------------------------------------------------------------------------------
%	12. LIMITATIONS (honesty)
%----------------------------------------------------------------------------------------

\begin{frame}
	\frametitle{Limitations}

	\begin{itemize}
		\item \textbf{Behaviour, not accuracy.} Reference outputs are the frozen models' own RGB predictions (pseudo-labels), not human annotations.
		\item \textbf{People untested.} Privacy-by-design excluded close-range faces, and \emph{person} is already among the weakest segmentation classes.
		\item \textbf{Geographically narrow.} Outdoor London only, one rig and one sensor.
		\item \textbf{Trained on sharp frames only.} Blurred and motion-corrupted captures were filtered out, so quality degrades on blurry or noisy input.
	\end{itemize}
\end{frame}

%----------------------------------------------------------------------------------------
%	13. CONCLUSION
%----------------------------------------------------------------------------------------

\begin{frame}
	\frametitle{Conclusion}

	\vspace{0.6\baselineskip}
	\begin{beamercolorbox}[wd=0.92\linewidth, sep=10pt, rounded=true]{keymsg}
		\centering Pixel fidelity does \textbf{not} predict downstream usefulness. What recovers behaviour
		is training and evaluating in the models' own feature spaces.
	\end{beamercolorbox}

	\vspace{0.6\baselineskip}
	Three contributions, new in combination:
	\begin{enumerate}
		\item A \textbf{shared-aperture rig} giving \textbf{12{,}536} simultaneously-aligned NIR/RGB pairs, which no public dataset offers.
		\item The \textbf{first use of NAFNet for NIR$\to$RGB}, with a \textbf{foundation-feature ensemble} replacing the perceptual loss.
		\item Distilled to a \textbf{1.7\,M} student ($67\times$ smaller) running real-time on the Pi: \textbf{5.05\,FPS} CPU, \textbf{27.2\,FPS} NPU.
	\end{enumerate}
\end{frame}

\begin{frame}
	\frametitle{Backup: why NAFNet?}
	\framesubtitle{Activation-free blocks $\Rightarrow$ predictable ranges $\Rightarrow$ clean INT8 / edge export}

	\begin{columns}[T]
		\begin{column}{0.52\linewidth}
			\centering
			\scriptsize
			\begin{tikzpicture}[
				op/.style={rectangle, rounded corners, draw=ICLBlue, fill=ICLBlue!8,
					minimum width=2.6cm, minimum height=0.5cm, align=center, font=\scriptsize},
				act/.style={rectangle, rounded corners, draw=BadRed, fill=BadRed!12,
					minimum width=2.6cm, minimum height=0.5cm, align=center, font=\scriptsize},
				gate/.style={rectangle, rounded corners, draw=GoodGreen, fill=GoodGreen!14,
					minimum width=2.6cm, minimum height=0.5cm, align=center, font=\scriptsize},
				ar/.style={-{Latex[length=1.4mm]}, ICLBlue}
			]
				% Standard block
				\node[font=\scriptsize\bfseries] (t1) at (0,0.7) {Standard conv block};
				\node[op]  (a1) at (0,0.1)  {$1{\times}1$ conv};
				\node[op]  (a2) at (0,-0.55){depthwise $3{\times}3$};
				\node[act] (a3) at (0,-1.2) {\textbf{ReLU / GELU}};
				\node[op]  (a4) at (0,-1.85){$1{\times}1$ conv};
				\draw[ar](a1)--(a2); \draw[ar](a2)--(a3); \draw[ar](a3)--(a4);
				% NAFBlock
				\node[font=\scriptsize\bfseries] (t2) at (3.4,0.7) {NAFBlock};
				\node[op]   (b1) at (3.4,0.1)  {$1{\times}1$ conv $C{\to}2C$};
				\node[op]   (b2) at (3.4,-0.55){depthwise $3{\times}3$};
				\node[gate] (b3) at (3.4,-1.2) {\textbf{SimpleGate}};
				\node[op]   (b4) at (3.4,-1.85){$1{\times}1$ conv};
				\draw[ar](b1)--(b2); \draw[ar](b2)--(b3); \draw[ar](b3)--(b4);
			\end{tikzpicture}

			\vspace{0.3\baselineskip}
			{\scriptsize SimpleGate: split channels in half, \textbf{multiply} the two halves. No nonlinear activation.}
		\end{column}
		\begin{column}{0.46\linewidth}
			\textbf{Why it matters here}
			\begin{itemize}
				\item No saturating activations $\Rightarrow$ \textbf{predictable dynamic range}, nothing to clip-calibrate for INT8.
				\item Small, standard operator set (conv, LayerNorm, multiply, PixelShuffle) $\Rightarrow$ \textbf{clean ONNX / LiteRT / Hailo} export.
				\item Zero-init residual scales $\Rightarrow$ stable training at batch size 1.
			\end{itemize}
			{\footnotesize Non-standard choice: restoration backbones target same-modality tasks, but here it infers \emph{missing} colour.}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{Backup: anticipated questions}

	\small
	\begin{itemize}
		\item \textbf{Pseudo-labels circular?} Independent probes (DINOv2-L, SigLIP2) were held out of training and still close $>0.5$.
		\item \textbf{Why detection weaker?} Small objects and fine texture reproduce least faithfully. Segmentation and depth lean on large-scale structure.
		\item \textbf{CLIP/DINOv2-S contaminated?} Yes, partially, so they are reported as \emph{supporting}, not primary, evidence.
		\item \textbf{Why not diffusion?} Iterative and stochastic, so it violates the latency budget and determinism (F3).
		\item \textbf{Why not fine-tune the downstream models?} Needs weights and labelled NIR per task, out of scope. The translator leaves them unchanged.
	\end{itemize}
\end{frame}
